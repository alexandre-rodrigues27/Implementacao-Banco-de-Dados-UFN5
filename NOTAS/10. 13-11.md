# Teste de Performance e Índices no SQL Server
## Introdução
Este projeto tem como objetivo analisar o desempenho de consultas no SQL Server utilizando uma tabela com mais de 300 mil registros. O teste envolve consultas com LIKE e igualdade (=), criação de índices não clusterizados e análise de desempenho utilizando STATISTICS IO e STATISTICS TIME.

## Código
```sql
CREATE TABLE Funcionario (
	Id INT PRIMARY KEY,
	Nome VARCHAR(100),
	Cargo VARCHAR(50),
	Salario DECIMAL(10,2),
	DataAdmissao DATE
);


INSERT INTO Funcionario
VALUES 
	(1, 'Jão', 'Adm', 3500, GETDATE()),
	(2, 'Alex', 'Estagiário', 450, GETDATE()),
	(3, 'Maria', 'TI', 2100, GETDATE()),
	(4, 'Rodrigo', 'Suporte', 1400, GETDATE()),
	(5, 'Carla', 'Limpeza', 1100, GETDATE());


DECLARE @contador INT = 1;
DECLARE @Id INT;

SELECT @Id = ISNULL (MAX(Id), 0) FROM Funcionario;

WHILE @contador <= 300000 
BEGIN
	SET @Id = @Id + 1;

	INSERT INTO Funcionario
	VALUES (
		@Id, 

		'Funcionário' + CAST (@Id AS VARCHAR (10)),

		CASE @contador % 5
			WHEN 0 THEN 'Gerente'
			WHEN 1 THEN 'Desenvolvedor'
			WHEN 2 THEN 'Analista'
			WHEN 3 THEN 'Suporte'
			ELSE 'Financeiro'
		END,

		FLOOR(10000 + RAND ()) + 1600,

		DATEADD(DAY, @contador % 3650, '20150101'))

	SET @contador = @contador + 1;
END
GO


SELECT * FROM Funcionario;
GO

SET STATISTICS IO ON;
SET STATISTICS TIME ON;

SELECT * 
FROM Funcionario
WHERE Nome LIKE '%255000';

SELECT * 
FROM Funcionario
WHERE Nome = 'Funcionário255000';


-- Criando INDICE --

CREATE NONCLUSTERED INDEX ix_Funcionario_Nome
ON Funcionario (Nome);

SELECT *
FROM Funcionario
WHERE DataAdmissao > '20150101' AND DataAdmissao <= '20150201';

DROP INDEX ix_Funcionario_Nome ON Funcionario;
```
# Importação de Dados JSON no SQL Server
## Introdução 
Este projeto tem como objetivo praticar a importação de grandes volumes de dados no SQL Server utilizando arquivos JSON. O processo envolve a criação de uma tabela, geração de dados fictícios através de um JSON Generator, leitura de arquivos externos com OPENROWSET e extração das informações utilizando OPENJSON para inserir os registros no banco de dados.

```sql
CREATE TABLE Pessoa (
    id INT PRIMARY KEY IDENTITY,
    nome VARCHAR(255),
    cidade VARCHAR(255),
    estado VARCHAR(255),
    email VARCHAR(255)
);


-- Popular Banco com JSON GENERATOR --

-- [
--  '{{repeat(200000, 250000)}}',
--  {
--    nome: '{{firstName()}} {{surname()}}',
--    cidade: '{{city()}}',
--    estado: '{{state()}}',
--    email: '{{email()}}'
--  }
-- ]

DECLARE @json NVARCHAR(MAX);

SELECT @json = BulkColumn
FROM OPENROWSET(
    BULK 'C:\Users\laboratorio\Downloads\generated.json',
    SINGLE_CLOB,
    CODEPAGE = '65001'
) AS j;

INSERT INTO Pessoa (nome, cidade, estado, email)
SELECT
    nome,
    cidade,
    estado,
    email
FROM OPENJSON(@json)
WITH (
    nome   VARCHAR(255) '$.nome',
    cidade VARCHAR(255) '$.cidade',
    estado VARCHAR(255) '$.estado',
    email  VARCHAR(255) '$.email'
);

SELECT * FROM Pessoa;
```
